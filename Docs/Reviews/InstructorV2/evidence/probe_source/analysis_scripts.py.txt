import csv, sys, math
def load(p):
    with open(p) as f: return list(csv.DictReader(f))
d = sys.argv[1]
def show(name, t0, t1, every, cols):
    rows = load(d + "/probe_" + name + ".csv")
    print("==", name, cols)
    for r in rows:
        t = float(r["t"])
        if t0 <= t <= t1 and abs((t / every) - round(t / every)) < 1e-6:
            print("  " + " ".join(f"{c}={r[c]}" for c in ["t"] + cols))
def extrema(name, col, t0):
    rows = load(d + "/probe_" + name + ".csv")
    ts=[float(r["t"]) for r in rows]; v=[float(r[col]) for r in rows]
    ex=[]
    for i in range(1,len(v)-1):
        if ts[i]>=t0 and ((v[i]>v[i-1] and v[i]>=v[i+1]) or (v[i]<v[i-1] and v[i]<=v[i+1])) and abs(v[i])>1:
            ex.append((ts[i],v[i]))
    print("==", name, col, "extrema after", t0, ":", " ".join(f"{t:.1f}:{x:.1f}" for t,x in ex[:30]))
